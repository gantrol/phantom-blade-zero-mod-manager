cmake_minimum_required(VERSION 3.25)
project(PhantomBladeArena LANGUAGES C CXX)
add_library(arena_hooks STATIC vendor/minhook/src/buffer.c vendor/minhook/src/hook.c
    vendor/minhook/src/trampoline.c vendor/minhook/src/hde/hde64.c)
target_include_directories(arena_hooks PUBLIC vendor/minhook/include)
set_property(TARGET arena_hooks PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
add_library(PhantomBladeArena SHARED src/main.cpp src/Reflection.cpp src/Menu.cpp src/Isolation.cpp src/Attributes.cpp src/Battle.cpp)
option(ARENA_DIAGNOSTICS "Include local development reflection console" OFF)
if(ARENA_DIAGNOSTICS)
    target_sources(PhantomBladeArena PRIVATE src/Diagnostics.cpp)
    target_compile_definitions(PhantomBladeArena PRIVATE ARENA_DIAGNOSTICS)
endif()
target_compile_features(PhantomBladeArena PRIVATE cxx_std_23)
target_include_directories(PhantomBladeArena PRIVATE include "${ARENA_GENERATED_INCLUDE}")
target_include_directories(PhantomBladeArena SYSTEM PRIVATE vendor/include)
target_compile_definitions(PhantomBladeArena PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN HAS_INPUT)
target_compile_options(PhantomBladeArena PRIVATE /utf-8 /W4 /permissive- /EHsc)
set_property(TARGET PhantomBladeArena PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
target_link_libraries(PhantomBladeArena PRIVATE "${ARENA_UE4SS_LIBRARY}" arena_hooks)
set_target_properties(PhantomBladeArena PROPERTIES OUTPUT_NAME main)
