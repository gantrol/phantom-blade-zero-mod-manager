cmake_minimum_required(VERSION 3.25)
project(PhantomBladeMapBridge LANGUAGES CXX)
add_library(PhantomBladeMapBridge SHARED src/PhantomBladeMap.Bridge/main.cpp)
target_compile_features(PhantomBladeMapBridge PRIVATE cxx_std_23)
set_property(TARGET PhantomBladeMapBridge PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
target_compile_definitions(PhantomBladeMapBridge PRIVATE HAS_INPUT)
target_include_directories(PhantomBladeMapBridge SYSTEM PRIVATE vendor/include)
target_compile_options(PhantomBladeMapBridge PRIVATE /utf-8 /W4 /permissive-)
target_link_libraries(PhantomBladeMapBridge PRIVATE "${PBM_IMPORT_LIBRARY}")
set_target_properties(PhantomBladeMapBridge PROPERTIES OUTPUT_NAME main)
